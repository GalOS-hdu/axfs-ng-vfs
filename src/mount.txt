//! 挂载点（Mountpoint）和位置（Location）相关实现
//! 负责管理文件系统的挂载、卸载以及路径解析等功能

use alloc::{
    string::String,          // 动态字符串类型
    sync::{Arc, Weak},       // 线程安全的引用计数（强引用、弱引用）
    vec,                     // 动态数组
};
use core::{
    iter,                    // 迭代器相关
    mem,                     // 内存操作相关
    sync::atomic::{AtomicU64, Ordering},  // 原子操作
    task::Context,           // 异步任务上下文
};

use axpoll::{IoEvents, Pollable};  // I/O事件轮询相关 trait
use hashbrown::HashMap;            // 高效哈希表实现
use inherit_methods_macro::inherit_methods;  // 方法继承宏

use crate::{
    DirEntry,                // 目录项（文件/目录的抽象）
    DirEntrySink,            // 目录项接收者 trait（用于读取目录）
    Filesystem,              // 文件系统封装结构
    FilesystemOps,           // 文件系统操作 trait
    Metadata,                // 文件元数据
    MetadataUpdate,          // 元数据更新指令
    Mutex,                   // 互斥锁（线程安全同步）
    MutexGuard,              // 互斥锁守卫（自动释放锁）
    NodeFlags,               // 节点标志（如流、不可缓存等）
    NodePermission,          // 节点权限（读写执行权限）
    NodeType,                // 节点类型（文件、目录、符号链接等）
    OpenOptions,             // 文件打开选项
    ReferenceKey,            // 目录项引用键（用于哈希表索引）
    TypeMap,                 // 类型安全的用户数据存储
    VfsError,                // VFS 错误类型
    VfsResult,               // VFS 操作结果类型
    path::{DOT, DOTDOT, PathBuf},  // 路径相关常量和结构（.、..、路径缓冲）
};

/// 挂载点结构体，表示一个文件系统被挂载到的位置
#[derive(Debug)]
pub struct Mountpoint {
    /// 挂载点的根目录项（对应被挂载文件系统的根）
    root: DirEntry,
    /// 在父挂载点中的位置（None 表示根挂载点）
    location: Option<Location>,
    /// 子挂载点集合（以子目录项的引用键为索引，存储弱引用避免循环引用）
    children: Mutex<HashMap<ReferenceKey, Weak<Self>>>,
    /// 设备 ID（唯一标识该挂载点对应的设备）
    device: u64,
}

impl Mountpoint {
    /// 创建新的挂载点
    ///
    /// # 参数
    /// * `fs` - 被挂载的文件系统
    /// * `location_in_parent` - 在父挂载点中的位置（父目录的 Location）
    ///
    /// # 返回值
    /// 新挂载点的 Arc 引用（线程安全）
    pub fn new(fs: &Filesystem, location_in_parent: Option<Location>) -> Arc<Self> {
        // 静态原子计数器，用于生成唯一设备 ID（从 1 开始）
        static DEVICE_COUNTER: AtomicU64 = AtomicU64::new(1);

        // 获取文件系统的根目录项作为挂载点的根
        let root = fs.root_dir();
        Arc::new(Self {
            root,
            location: location_in_parent,
            children: Mutex::default(),  // 初始化空的子挂载点哈希表
            device: DEVICE_COUNTER.fetch_add(1, Ordering::Relaxed),  // 原子递增并获取设备 ID
        })
    }

    /// 创建根挂载点（无父挂载点）
    ///
    /// # 参数
    /// * `fs` - 根文件系统
    ///
    /// # 返回值
    /// 根挂载点的 Arc 引用
    pub fn new_root(fs: &Filesystem) -> Arc<Self> {
        Self::new(fs, None)  // 根挂载点的 location 为 None
    }

    /// 获取当前挂载点的根位置（根目录项对应的 Location）
    ///
    /// # 返回值
    /// 由当前挂载点和其根目录项组成的 Location
    pub fn root_location(self: &Arc<Self>) -> Location {
        Location::new(self.clone(), self.root.clone())
    }

    /// 获取在父挂载点中的位置
    ///
    /// # 返回值
    /// 父位置的克隆（Option<Location>）
    pub fn location(&self) -> Option<Location> {
        self.location.clone()
    }

    /// 判断当前挂载点是否为根挂载点
    ///
    /// # 返回值
    /// 若 location 为 None 则返回 true，否则为 false
    pub fn is_root(&self) -> bool {
        self.location.is_none()
    }

    /// 获取有效的挂载点（处理挂载覆盖场景）
    ///
    /// 例如：先将 /dev/sda1 挂载到 /mnt，再将 /dev/sda2 挂载到 /mnt，
    /// 此时 /mnt 实际指向 sda2 的挂载点，该方法返回 sda2 的挂载点。
    ///
    /// # 返回值
    /// 最终有效的挂载点 Arc 引用
    pub(crate) fn effective_mountpoint(self: &Arc<Self>) -> Arc<Mountpoint> {
        let mut mountpoint = self.clone();
        // 循环查找：如果当前挂载点的根目录项上有子挂载点，则更新为子挂载点
        while let Some(mount) = mountpoint.root.as_dir().unwrap().mountpoint() {
            mountpoint = mount;
        }
        mountpoint
    }

    /// 获取当前挂载点的设备 ID
    ///
    /// # 返回值
    /// 设备 ID（u64）
    pub fn device(self: &Arc<Self>) -> u64 {
        self.device
    }
}

/// 位置结构体，表示文件系统中一个具体的路径位置（挂载点 + 目录项）
#[derive(Debug, Clone)]
pub struct Location {
    /// 该位置所属的挂载点
    mountpoint: Arc<Mountpoint>,
    /// 对应的目录项（文件/目录）
    entry: DirEntry,
}

/// 从 self.entry 继承方法（通过宏自动生成）
/// 这些方法直接调用 entry 的对应实现，避免代码重复
#[inherit_methods(from = "self.entry")]
impl Location {
    /// 获取目录项的 inode 编号
    pub fn inode(&self) -> u64;

    /// 获取所属的文件系统操作接口
    pub fn filesystem(&self) -> &dyn FilesystemOps;

    /// 更新目录项的元数据
    pub fn update_metadata(&self, update: MetadataUpdate) -> VfsResult<()>;

    /// 获取目录项的大小（字节数）
    #[allow(clippy::len_without_is_empty)]
    pub fn len(&self) -> VfsResult<u64>;

    /// 同步目录项数据到存储设备
    pub fn sync(&self, data_only: bool) -> VfsResult<()>;

    /// 判断是否为文件
    pub fn is_file(&self) -> bool;

    /// 判断是否为目录
    pub fn is_dir(&self) -> bool;

    /// 获取节点类型（文件/目录/符号链接等）
    pub fn node_type(&self) -> NodeType;

    /// 判断是否为挂载点的根目录项
    pub fn is_root_of_mount(&self) -> bool;

    /// 读取符号链接的目标路径
    pub fn read_link(&self) -> VfsResult<String>;

    /// 执行 IO 控制命令
    pub fn ioctl(&self, cmd: u32, arg: usize) -> VfsResult<usize>;

    /// 获取节点标志
    pub fn flags(&self) -> NodeFlags;

    /// 获取用户数据存储（类型安全的键值对）
    pub fn user_data(&self) -> MutexGuard<'_, TypeMap>;
}

impl Location {
    /// 创建新的位置实例
    ///
    /// # 参数
    /// * `mountpoint` - 所属的挂载点
    /// * `entry` - 对应的目录项
    ///
    /// # 返回值
    /// 新的 Location 实例
    pub fn new(mountpoint: Arc<Mountpoint>, entry: DirEntry) -> Self {
        Self { mountpoint, entry }
    }

    /// 用当前挂载点包装一个新的目录项，生成新的 Location
    ///
    /// # 参数
    /// * `entry` - 新的目录项
    ///
    /// # 返回值
    /// 新的 Location 实例（挂载点不变，目录项更新）
    fn wrap(&self, entry: DirEntry) -> Self {
        Self::new(self.mountpoint.clone(), entry)
    }

    /// 获取所属的挂载点（不可变引用）
    pub fn mountpoint(&self) -> &Arc<Mountpoint> {
        &self.mountpoint
    }

    /// 获取对应的目录项（不可变引用）
    pub fn entry(&self) -> &DirEntry {
        &self.entry
    }

    /// 获取当前位置的名称（文件名或目录名）
    ///
    /// 特殊处理：若为挂载点的根目录项，则名称继承自父挂载点中该挂载点的名称
    pub fn name(&self) -> &str {
        if self.is_root_of_mount() {
            // 挂载点的根目录项名称 = 父挂载点中该挂载点的名称
            self.mountpoint.location.as_ref().map_or("", Location::name)
        } else {
            // 普通目录项直接返回自身名称
            self.entry.name()
        }
    }

    /// 获取父目录的位置
    ///
    /// # 返回值
    /// 父目录的 Location（若为挂载点的根目录项，则返回父挂载点的父目录）
    pub fn parent(&self) -> Option<Self> {
        if !self.is_root_of_mount() {
            // 非挂载点根目录项：直接取自身的父目录项
            return Some(self.wrap(self.entry.parent().unwrap()));
        }
        // 挂载点根目录项：父目录为父挂载点的位置的父目录
        self.mountpoint.location()?.parent()
    }

    /// 判断当前位置是否为整个 VFS 的根（根挂载点的根目录项）
    pub fn is_root(&self) -> bool {
        self.mountpoint.is_root() && self.entry.is_root_of_mount()
    }

    /// 检查当前位置是否为目录，非目录则返回错误
    pub fn check_is_dir(&self) -> VfsResult<()> {
        self.entry.as_dir().map(|_| ())  // 调用 entry 的 as_dir，成功则返回 ()
    }

    /// 检查当前位置是否为文件，非文件则返回错误
    pub fn check_is_file(&self) -> VfsResult<()> {
        self.entry.as_file().map(|_| ())  // 调用 entry 的 as_file，成功则返回 ()
    }

    /// 获取当前位置的元数据（包含设备 ID）
    ///
    /// # 返回值
    /// 包含设备 ID 的 Metadata
    pub fn metadata(&self) -> VfsResult<Metadata> {
        let mut metadata = self.entry.metadata()?;  // 获取目录项的元数据
        metadata.device = self.mountpoint.device();  // 补充当前挂载点的设备 ID
        Ok(metadata)
    }

    /// 获取绝对路径
    ///
    /// # 返回值
    /// 完整的绝对路径 PathBuf
    pub fn absolute_path(&self) -> VfsResult<PathBuf> {
        let mut components = vec![];  // 存储路径组件（从当前目录向上收集）
        let mut cur = self.clone();
        loop {
            // 收集当前目录项的路径组件
            cur.entry.collect_absolute_path(&mut components);
            // 向上追溯到父挂载点的位置
            cur = match cur.mountpoint.location() {
                Some(loc) => loc,
                None => break,  // 到达根挂载点，终止循环
            }
        }
        // 拼接路径：根目录 + 反转后的组件（因为是从当前目录向上收集的）
        Ok(iter::once("/")
            .chain(components.iter().map(String::as_str).rev())
            .collect())
    }

    /// 比较两个 Location 是否指向同一个内存地址（指针相等）
    pub fn ptr_eq(&self, other: &Self) -> bool {
        Arc::ptr_eq(&self.mountpoint, &other.mountpoint) && self.entry.ptr_eq(&other.entry)
    }

    /// 判断当前位置是否是一个挂载点（目录项上挂载了其他文件系统）
    pub fn is_mountpoint(&self) -> bool {
        self.entry.as_dir().is_ok_and(|it| it.is_mountpoint())
    }

    /// 解析挂载点（若当前目录项上有挂载点，则返回该挂载点的根位置）
    /// 参见 [`Mountpoint::effective_mountpoint`]
    fn resolve_mountpoint(self) -> Self {
        // 若当前目录项是目录且有挂载点，则返回该挂载点的根位置
        let Some(mountpoint) = self.entry.as_dir().ok().and_then(|it| it.mountpoint()) else {
            return self;  // 非目录或无挂载点，返回自身
        };
        let mountpoint = mountpoint.effective_mountpoint();  // 获取有效的挂载点
        let entry = mountpoint.root.clone();  // 挂载点的根目录项
        Self::new(mountpoint, entry)
    }

    /// 查找目录项（不跟随符号链接）
    ///
    /// # 参数
    /// * `name` - 要查找的名称（支持 . 和 ..）
    ///
    /// # 返回值
    /// 找到的目录项对应的 Location
    pub fn lookup_no_follow(&self, name: &str) -> VfsResult<Self> {
        Ok(match name {
            DOT => self.clone(),  // . 表示当前目录
            DOTDOT => self.parent().unwrap_or_else(|| self.clone()),  // .. 表示父目录（无父则返回自身）
            _ => {
                // 查找子目录项，并解析可能的挂载点
                let loc = Self::new(self.mountpoint.clone(), self.entry.as_dir()?.lookup(name)?);
                loc.resolve_mountpoint()
            }
        })
    }

    /// 在当前目录下创建新的节点（文件/目录/符号链接等）
    ///
    /// # 参数
    /// * `name` - 新节点的名称
    /// * `node_type` - 节点类型
    /// * `permission` - 节点权限
    ///
    /// # 返回值
    /// 新节点的 Location
    pub fn create(
        &self,
        name: &str,
        node_type: NodeType,
        permission: NodePermission,
    ) -> VfsResult<Self> {
        self.entry
            .as_dir()?  // 确保当前是目录
            .create(name, node_type, permission)  // 创建节点
            .map(|entry| self.wrap(entry))  // 包装为 Location
    }

    /// 创建硬链接
    ///
    /// # 参数
    /// * `name` - 链接名称
    /// * `node` - 被链接的节点
    ///
    /// # 返回值
    /// 新链接的 Location
    ///
    /// # 错误
    /// 若跨设备（不同挂载点）则返回 CrossesDevices 错误
    pub fn link(&self, name: &str, node: &Self) -> VfsResult<Self> {
        // 硬链接不能跨设备（必须在同一挂载点）
        if !Arc::ptr_eq(&self.mountpoint, &node.mountpoint) {
            return Err(VfsError::CrossesDevices);
        }
        self.entry
            .as_dir()?  // 确保当前是目录
            .link(name, &node.entry)  // 创建链接
            .map(|entry| self.wrap(entry))  // 包装为 Location
    }

    /// 重命名文件或目录
    ///
    /// # 参数
    /// * `src_name` - 源名称
    /// * `dst_dir` - 目标目录
    /// * `dst_name` - 目标名称
    ///
    /// # 错误
    /// * 跨设备：返回 CrossesDevices
    /// * 源是目标的祖先目录：返回 InvalidInput
    pub fn rename(&self, src_name: &str, dst_dir: &Self, dst_name: &str) -> VfsResult<()> {
        // 重命名不能跨设备
        if !Arc::ptr_eq(&self.mountpoint, &dst_dir.mountpoint) {
            return Err(VfsError::CrossesDevices);
        }
        // 不能将目录重命名到其子目录中（避免循环）
        if !self.ptr_eq(dst_dir) && self.entry.is_ancestor_of(&dst_dir.entry)? {
            return Err(VfsError::InvalidInput);
        }
        self.entry
            .as_dir()?  // 源所在目录
            .rename(src_name, dst_dir.entry.as_dir()?, dst_name)  // 执行重命名
    }

    /// 删除目录项（文件或目录）
    ///
    /// # 参数
    /// * `name` - 要删除的名称
    /// * `is_dir` - 是否为目录（用于校验）
    pub fn unlink(&self, name: &str, is_dir: bool) -> VfsResult<()> {
        self.entry.as_dir()?.unlink(name, is_dir)
    }

    /// 打开文件（在当前目录下查找并打开）
    ///
    /// # 参数
    /// * `name` - 文件名
    /// * `options` - 打开选项
    ///
    /// # 返回值
    /// 打开的文件对应的 Location
    pub fn open_file(&self, name: &str, options: &OpenOptions) -> VfsResult<Location> {
        self.entry
            .as_dir()?  // 确保当前是目录
            .open_file(name, options)  // 打开文件
            .map(|entry| self.wrap(entry).resolve_mountpoint())  // 解析挂载点并包装
    }

    /// 读取目录项
    ///
    /// # 参数
    /// * `offset` - 读取偏移量（用于分页读取）
    /// * `sink` - 目录项接收者（用于收集结果）
    ///
    /// # 返回值
    /// 读取的目录项数量
    pub fn read_dir(&self, offset: u64, sink: &mut dyn DirEntrySink) -> VfsResult<usize> {
        self.entry.as_dir()?.read_dir(offset, sink)
    }

    /// 挂载文件系统到当前位置（必须是目录）
    ///
    /// # 参数
    /// * `fs` - 要挂载的文件系统
    ///
    /// # 返回值
    /// 新创建的挂载点 Arc 引用
    ///
    /// # 错误
    /// 若当前位置已挂载文件系统，则返回 ResourceBusy
    pub fn mount(&self, fs: &Filesystem) -> VfsResult<Arc<Mountpoint>> {
        // 锁定当前目录项的挂载点字段（避免并发修改）
        let mut mountpoint = self.entry.as_dir()?.mountpoint.lock();
        if mountpoint.is_some() {
            // 已挂载其他文件系统，返回忙错误
            return Err(VfsError::ResourceBusy);
        }
        // 创建新挂载点（父位置为当前 Location）
        let result = Mountpoint::new(fs, Some(self.clone()));
        *mountpoint = Some(result.clone());  // 记录挂载点到目录项
        // 将新挂载点添加到父挂载点的子挂载点集合（弱引用）
        self.mountpoint
            .children
            .lock()
            .insert(self.entry.key(), Arc::downgrade(&result));
        Ok(result)
    }

    /// 卸载当前位置的文件系统（必须是挂载点的根目录）
    ///
    /// # 错误
    /// * 非挂载点的根目录：返回 InvalidInput
    /// * 存在子挂载点：返回 ResourceBusy
    pub fn unmount(&self) -> VfsResult<()> {
        // 必须是挂载点的根目录才能卸载
        if !self.is_root_of_mount() {
            return Err(VfsError::InvalidInput);
        }
        // 存在子挂载点则不允许卸载
        if !self.mountpoint.children.lock().is_empty() {
            return Err(VfsError::ResourceBusy);
        }
        // 校验：当前目录项必须是挂载点的根目录项
        assert!(self.entry.ptr_eq(&self.mountpoint.root));
        // 清理目录项（如缓存等）
        self.entry.as_dir()?.forget();
        // 从父挂载点中移除当前挂载点的记录
        if let Some(parent_loc) = &self.mountpoint.location {
            *parent_loc.entry.as_dir()?.mountpoint.lock() = None;
        }
        Ok(())
    }

    /// 卸载当前挂载点及其所有子挂载点
    ///
    /// # 错误
    /// 非挂载点的根目录：返回 InvalidInput
    pub fn unmount_all(&self) -> VfsResult<()> {
        // 必须是挂载点的根目录
        if !self.is_root_of_mount() {
            return Err(VfsError::InvalidInput);
        }
        // 取出所有子挂载点（转移所有权，避免锁定期间修改）
        let children = mem::take(&mut *self.mountpoint.children.lock());
        // 递归卸载所有子挂载点
        for (_, child) in children {
            if let Some(child) = child.upgrade() {  // 弱引用升级为强引用
                child.root_location().unmount_all()?;  // 递归卸载子挂载点
            }
        }
        // 最后卸载自身
        self.unmount()
    }
}

/// 为 Location 实现 Pollable trait（继承自 entry）
/// 用于 I/O 事件轮询（如可读、可写事件）
#[inherit_methods(from = "self.entry")]
impl Pollable for Location {
    /// 检查当前可触发的 I/O 事件
    fn poll(&self) -> IoEvents;

    /// 注册 I/O 事件监听（关联异步任务上下文）
    fn register(&self, context: &mut Context<'_>, events: IoEvents);
}
